{
		Name: "stashing working set with deleted table and popping it",
		SetUpScript: []string{

					{"new_tab"},
				},
			},
			{
				Query:            "CALL DOLT_STASH('pop', 'myStash');",
				SkipResultsCheck: true,
			},

			},
